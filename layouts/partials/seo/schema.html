{{- $scr := .Scratch -}} {{- $page := . -}} {{- if eq .Type "blog" -}} {{- $scr.Set "schema_type" "Article" -}} {{- else if eq .Type "contact" -}} {{- $scr.Set "schema_type" "ContactPage" -}} {{- else -}} {{- $scr.Set "schema_type" "WebPage" -}} {{- end -}} {{- $title := .Title -}} {{- if .IsHome -}} {{- $title = .Site.Title -}} {{- end -}} {{- $description := .Description -}} {{- if not $description -}} {{- $description = .Site.Params.metadata.description -}} {{- end -}} {{- if .IsHome -}} {{- /* Homepage: WebSite schema with SearchAction */ -}} {{- else if eq .Type "blog" -}} {{- /* Blog posts: Article schema with enhanced metadata. Author resolution: front-matter `author` is treated as a slug lookup against site.Data.authors. If the slug exists, emit a schema.org Person with credentials and sameAs (E-E-A-T signal that Gemini 3 and Claude weight heavily). Otherwise fall back to the Organization author for legacy posts. */ -}} {{- $authorSlug := .Params.author | default "" -}} {{- $authorKey := $authorSlug | urlize -}} {{- $author := index site.Data.authors $authorKey -}} {{- else -}} {{- /* Other pages: WebPage/ContactPage schema */ -}} {{- end -}} {{- /* Organization Schema for all pages. sameAs is the entity-disambiguation surface LLMs use (Wikidata, Wikipedia, LinkedIn, Crunchbase, etc.). The wider and more cross-referenced this graph is, the more reliably ChatGPT, Gemini, Claude and Perplexity recognise TeamWheels as a single entity rather than conflating it with competitors. Profile URLs that don't yet exist are commented out — uncomment as each one is created. */ -}} {{- /* SoftwareApplication Schema - product rich results */ -}} {{- if or .IsHome (eq .Layout "pricing") (eq .Layout "how-it-works") -}} {{- end -}} {{- /* FAQPage Schema — emits on ANY page that declares an FAQ in front matter, not just /pricing/. Supports two front-matter shapes: 1. Pricing-style (enable + faq_list with title/content): faq: enable: true faq_list: - title: "Question?" content: "Answer." 2. Lightweight per-page list (question/answer): faq: - question: "Question?" answer: "Answer." Q&A blocks are the format LLMs preferentially extract — Search Engine Land found 72.4% of pages cited by ChatGPT contain a short, direct answer immediately after a question heading. */ -}} {{- $faqEntries := slice -}} {{- with .Params.faq -}} {{- if reflect.IsMap . -}} {{- if and .enable .faq_list -}} {{- range .faq_list -}} {{- $faqEntries = $faqEntries | append (dict "question" (or .title .question) "answer" (or .content .answer)) -}} {{- end -}} {{- end -}} {{- else if reflect.IsSlice . -}} {{- range . -}} {{- $q := or (index . "question") (index . "title") -}} {{- $a := or (index . "answer") (index . "content") -}} {{- if and $q $a -}} {{- $faqEntries = $faqEntries | append (dict "question" $q "answer" $a) -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- if $faqEntries -}} {{- end -}} {{- /* HowTo Schema — emits when a page declares a `howto:` block in front matter. Front-matter shape: howto: name: "Title of the procedure" description: "What this guide accomplishes" totalTime: "P30D" # ISO 8601 duration (optional) image: "images/foo.png" # optional supply: ["item one", ...] # optional, schema.org HowToSupply tool: ["Microsoft Teams"] # optional, schema.org HowToTool step: - name: "Step name" text: "What to do at this step" url: "#anchor" # optional, deep link into the article image: "images/x.png" # optional Required: name + step[] with name & text. */ -}} {{- with .Params.howto -}} {{- if and .name .step -}} {{- end -}} {{- end -}} {{- /* BreadcrumbList Schema — skip on homepage (single-item trail has no value) */ -}} {{- if not .IsHome -}} {{- $breadcrumbs := slice -}} {{- $breadcrumbs = $breadcrumbs | append (dict "position" 1 "name" "Home" "url" .Site.Home.Permalink) -}} {{- with .Section -}} {{- $sectionPage := site.GetPage . -}} {{- if $sectionPage -}} {{- $breadcrumbs = $breadcrumbs | append (dict "position" 2 "name" $sectionPage.Title "url" $sectionPage.Permalink) -}} {{- end -}} {{- end -}} {{- $breadcrumbs = $breadcrumbs | append (dict "position" (add (len $breadcrumbs) 1) "name" .Title "url" .Permalink) -}} {{- end -}}
