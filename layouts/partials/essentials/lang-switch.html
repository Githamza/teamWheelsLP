{{- /* Flag-based language switcher. Renders every site language as a flag + code link so the current/available languages are obvious at a glance. Links to the translated page when it exists, otherwise to the target language home. Pass the current page as context. */ -}} {{- $page := . -}} {{- $cur := .Lang -}}
{{- range hugo.Sites -}} {{- $l := .Language -}} {{- $url := "" -}} {{- range $page.AllTranslations -}} {{- if eq .Lang $l.Lang -}}{{- $url = .Permalink -}}{{- end -}} {{- end -}} {{- if not $url -}}{{- $url = printf "/%s/" $l.Lang | relURL -}}{{- end -}} {{- $active := eq $l.Lang $cur -}} {{ partial "essentials/flag.html" $l.Lang }} {{ $l.Lang | upper }} {{- end -}}
