{{- /* Flag-based language switcher. Renders every live site language as a flag + code link so the current/available languages are obvious at a glance. Links to the translated page when it exists, otherwise to the target language home. Pass the current page as context. A locale whose home page is still a draft is skipped (partials/lang-live.html). Regional locales (nl-be, nl) show the bare language code with their country flag; the full label goes in title and aria-label. */ -}} {{- $page := . -}} {{- $cur := .Lang -}}
{{- range hugo.Sites -}} {{- if partialCached "lang-live.html" . .Language.Lang -}} {{- $l := .Language -}} {{- $tag := .Params.hreflang -}} {{- $url := "" -}} {{- range $page.AllTranslations -}} {{- if eq .Lang $l.Lang -}}{{- $url = .Permalink -}}{{- end -}} {{- end -}} {{- if not $url -}}{{- $url = printf "/%s/" $l.Lang | relURL -}}{{- end -}} {{- $active := eq $l.Lang $cur -}} {{ partial "essentials/flag.html" $l.Lang }} {{ index (split $l.Lang "-") 0 | upper }} {{- end -}} {{- end -}}
